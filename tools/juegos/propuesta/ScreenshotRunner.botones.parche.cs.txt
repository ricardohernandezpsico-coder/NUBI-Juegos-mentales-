private static IEnumerator BotonesTomas()
{
    if (!NeuroVida.Bridge.EditorTools.PropuestaHook.Hay() || !NeuroVida.Bridge.EditorTools.PropuestaHook.CargaNivel(12)) yield break;
    yield return Wait.For(2.2);
    ShotRaw("rep");                                                                                                                  // reposo
    NeuroVida.Bridge.EditorTools.PropuestaHook.PresionaArrancar(); yield return Wait.For(0.03); ShotRaw("pres");                    // «Arrancar» presionado
    yield return Wait.For(0.5);
    NeuroVida.Bridge.EditorTools.PropuestaHook.Bloquea(); yield return Wait.For(0.5); ShotRaw("llaves");                            // las dos llaves usadas
}
